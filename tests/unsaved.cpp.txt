namespace SlitYaX86 {
struct Lyss {
private:
    std::vector<void*>    tofree_list;
public:
    _1Dlsd          *lsd;
    Binary          &base_img;

    std::map<__u64, struct _1Dlsd_entry>    map;
    Lyss&           self{*this};

    Lyss(_1Dlsd *l, Binary &_pe);
    virtual ~Lyss();
    Generic(T) void push(Binary *img, enum _1DLSD_TYPE t, T &v) {
        map[v.off] = (struct _1Dlsd_entry){
            .t = t, .mm = &v.mm, .src = {img, v.off, v.mm.sz}};
    }
    Generic(Fn) int feach(enum _1DLSD_TYPE &t, Fn fn) {
        return feach(t, nullptr, fn);
    }
    Generic(Fn) int feach(enum _1DLSD_TYPE &t, struct _1Dlsd_entry *v, Fn fn) {
        int i = 0;
        try {
            for (auto &[_, e] : map)
                if (e.t == t && ++i && !!e.mm) {
                    if (!!v) *v = e;
                    for (__u64 off = 0, sz = get_size(t); off < e.mm->sz; off += sz) {
                        fn(static_cast<lambda_arg<decltype(fn)>>(e.mm->mem+off));
                    }
                }
        } catch (...) {}

        return i;
    }
    std::map<__u64, struct _1Dlsd_entry> &operator()();
    virtual __u64 get_size(enum _1DLSD_TYPE &t) = default;
    bin_hdr_t *&operator<<(bin_hdr_t&& v);

//     void AlignHeaders(__u64 off, __u64 sz);
// private:
//     void ChopSections();
//     void ChopSegments();
};

std::map<__u64, struct _1Dlsd_entry> &operator()() {
    return map;
}


class WinLyss : public Lyss {
public:
    __u64 get_size(enum _1DLSD_TYPE &t) override;
}


bin_hdr_t *&WinLyss::operator<<(bin_hdr_t&& v) {
        auto v_ptr = new bin_hdr_t;
        tofree_list.push_back(reinterpret_cast<void*>(v_ptr));
        *v_ptr = std::move(v);
        return reinterpret_cast<bin_hdr_t*&>(tofree_list.back());
    }
__u64 WinLyss::get_size(enum _1DLSD_TYPE &t) override {
    __u64 v = 0;
    switch (t) {
        case DosHdr: v = sizeof(IMG_DOS_HEADER);     break;
        case NtHdrs: v = sizeof(IMG_NT_HEADERS);     break;
        case DataDir:v = sizeof(IMG_DATA_DIRECTORY); break;
        case Section:v = sizeof(IMG_SECTION_HEADER); break;
    }
    return v;
}
